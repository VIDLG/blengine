    pub fn begin_connect(&self, peripheral: PeripheralId,) -> Result<Arc<ConnectOperation>, BlengError>;
    pub fn begin_connect_with_options(&self, peripheral: PeripheralId, auto_connect: bool,) -> Result<Arc<ConnectOperation>, BlengError>;
    pub fn begin_disconnect(&self, connection: ConnectionId,) -> Result<Arc<DisconnectOperation>, BlengError>;
    pub fn begin_discover_gatt(&self, connection: ConnectionId,) -> Result<Arc<DiscoverGattOperation>, BlengError>;
    pub fn begin_read_characteristic(&self, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId,) -> Result<Arc<ReadOperation>, BlengError>;
    pub fn begin_write_characteristic(&self, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId, value: Vec<u8>, write_type: WriteType,) -> Result<Arc<WriteOperation>, BlengError>;
    pub fn begin_read_descriptor(&self, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId, descriptor: GattDescriptorId,) -> Result<Arc<ReadDescriptorOperation>, BlengError>;
    pub fn begin_write_descriptor(&self, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId, descriptor: GattDescriptorId, value: Vec<u8>,) -> Result<Arc<WriteDescriptorOperation>, BlengError>;
    pub fn begin_enable_notifications(&self, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId, mode: NotificationModeRequest,) -> Result<Arc<EnableNotificationsOperation>, BlengError>;
    pub fn begin_disable_notifications(&self, connection: ConnectionId, session: GattSessionId, subscription: SubscriptionId,) -> Result<Arc<DisableNotificationsOperation>, BlengError>;
    pub fn begin_read_rssi(&self, connection: ConnectionId,) -> Result<Arc<ReadRssiOperation>, BlengError>;
    pub fn begin_request_mtu(&self, connection: ConnectionId, mtu: u16,) -> Result<Arc<MtuOperation>, BlengError>;
    pub fn begin_query_mtu(&self, connection: ConnectionId,) -> Result<Arc<MtuOperation>, BlengError>;
    pub fn begin_set_connection_priority(&self, connection: ConnectionId, priority: ConnectionPriority,) -> Result<Arc<SetConnectionPriorityOperation>, BlengError>;
    pub fn begin_set_adapter_enabled(&self, enabled: bool,) -> Result<Arc<SetAdapterEnabledOperation>, BlengError>;
    pub fn begin_query_devices(&self, kind: DeviceQueryKind, service_uuids: Vec<BluetoothUuid>,) -> Result<Arc<QueryDevicesOperation>, BlengError>;
    pub fn begin_associate_companion(&self, service_uuids: Vec<BluetoothUuid>, display_name: Option<String>, device_profile: Option<String>, notify_on_device_nearby: bool,) -> Result<Arc<AssociateCompanionDeviceOperation>, BlengError>;
    pub fn begin_create_bond(&self, connection: ConnectionId, pin: Option<String>,) -> Result<Arc<BondOperation>, BlengError>;
    pub fn begin_remove_bond(&self, connection: ConnectionId,) -> Result<Arc<BondOperation>, BlengError>;
    pub fn begin_refresh_gatt_cache(&self, connection: ConnectionId,) -> Result<Arc<RefreshGattCacheOperation>, BlengError>;
    pub async fn initialize(&self, timeout_ms: u64) -> Result<AdapterSnapshot, BlengError>;
    pub async fn start_scan(&self, options: ScanOptions, timeout_ms: u64,) -> Result<ScanSessionId, BlengError>;
    pub async fn stop_scan(&self, session: ScanSessionId, timeout_ms: u64,) -> Result<(), BlengError>;
    pub async fn connect(&self, peripheral: PeripheralId, timeout_ms: u64,) -> Result<ConnectionId, BlengError>;
    pub async fn connect_with_options(&self, peripheral: PeripheralId, auto_connect: bool, timeout_ms: u64,) -> Result<ConnectionId, BlengError>;
    pub async fn set_adapter_enabled(&self, enabled: bool, timeout_ms: u64,) -> Result<(), BlengError>;
    pub async fn query_devices(&self, kind: DeviceQueryKind, service_uuids: Vec<BluetoothUuid>, timeout_ms: u64,) -> Result<Vec<DeviceRecord>, BlengError>;
    pub async fn disconnect(&self, connection: ConnectionId, timeout_ms: u64,) -> Result<(), BlengError>;
    pub async fn discover_gatt(&self, connection: ConnectionId, timeout_ms: u64,) -> Result<GattDatabase, BlengError>;
    pub async fn read_characteristic(&self, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId, timeout_ms: u64,) -> Result<Vec<u8>, BlengError>;
    pub async fn write_characteristic(&self, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId, value: Vec<u8>, write_type: WriteType, timeout_ms: u64,) -> Result<(), BlengError>;
    pub async fn enable_notifications(&self, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId, mode: NotificationModeRequest, timeout_ms: u64,) -> Result<NotificationSubscription, BlengError>;
    pub async fn disable_notifications(&self, connection: ConnectionId, session: GattSessionId, subscription: SubscriptionId, timeout_ms: u64,) -> Result<(), BlengError>;
    pub async fn read_rssi(&self, connection: ConnectionId, timeout_ms: u64,) -> Result<i16, BlengError>;
    pub async fn request_mtu(&self, connection: ConnectionId, mtu: u16, timeout_ms: u64,) -> Result<MtuInfo, BlengError>;
    pub async fn query_mtu(&self, connection: ConnectionId, timeout_ms: u64,) -> Result<MtuInfo, BlengError>;
    pub async fn read_descriptor(&self, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId, descriptor: GattDescriptorId, timeout_ms: u64,) -> Result<Vec<u8>, BlengError>;
    pub async fn write_descriptor(&self, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId, descriptor: GattDescriptorId, value: Vec<u8>, timeout_ms: u64,) -> Result<(), BlengError>;
    pub async fn set_connection_priority(&self, connection: ConnectionId, priority: ConnectionPriority, timeout_ms: u64,) -> Result<(), BlengError>;
    pub async fn create_bond(&self, connection: ConnectionId, timeout_ms: u64,) -> Result<BondState, BlengError>;
    pub async fn remove_bond(&self, connection: ConnectionId, timeout_ms: u64,) -> Result<BondState, BlengError>;
    pub async fn refresh_gatt_cache(&self, connection: ConnectionId, timeout_ms: u64,) -> Result<(), BlengError>;
    pub async fn next_event(&self, timeout_ms: u64) -> Result<Option<BleEvent>, BlengError>;
    pub async fn shutdown(&self) -> Result<(), BlengError>;
}

pub enum BlengBusyReason {
    Scan,
    Peripheral,
    Gatt,
    Operation,
    BackendSlotOccupied,
}

pub enum BlengContractFailure {
    InvalidCommandId,
    DuplicateHubOwnership,
}

pub enum BlengError {
    Closed,
    Cancelled,
    OperationNotFound {command_id: CommandId},
    OperationAlreadyCompleted {command_id: CommandId},
    OperationNotCancellable {command_id: CommandId},
    OperationResultConsumed,
    Disconnected,
    InvalidTimeout,
    NotReady {reason: BlengNotReadyReason},
    Busy {reason: BlengBusyReason},
    ResourceExhausted {resource: BlengResource},
    ContractFailure {reason: BlengContractFailure},
    DispatchRejected {error: CommandDispatchError},
    Operation {error: BleEventError},
    Assembly {error: BleHubError},
    Cleanup {error: BleHubError},
    Timeout,
    TransportFault,
    WorkerStopped,
}

pub enum BlengNotReadyReason {
    BackendNotAssembled,
    NotInitialized,
    ScanNotActive,
    ConnectionNotActive,
    GattNotReady,
    SubscriptionNotActive,
}

pub enum BlengResource {
    PendingWaiters,
    SubmissionQueue,
    Identifiers,
}
