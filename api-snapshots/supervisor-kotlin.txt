interface SupervisorInterface {
    fun activeContext(): ForeignOwnerContext?
    fun activeOwnerStart(): ForeignOwnerStart?
    fun configureOldOperationBarrier(oldLink: ForeignRecoveryLink?, expectedOldOperationIds: List<String>)
    suspend fun createFreshOwner(context: ForeignOwnerContext): ForeignRecoveryPlan
    suspend fun drivePlatformActionsUntilBoundary(): ForeignRecoveryPlan
    suspend fun executePlatformAction(): ForeignRecoveryPlan
    fun isActive(): Boolean
    fun lastRetiredResources(): List<ForeignResourceCount>?
    fun observe(observation: ForeignRecoveryObservation): ForeignRecoveryPlan
    fun observeExternalSignal(signal: ForeignExternalRecoverySignal): ForeignRecoveryPlan
    fun observeSignal(signal: ForeignRecoverySignal): ForeignRecoveryPlan
    suspend fun retireForRecovery(): ForeignRecoveryPlan
    suspend fun shutdown(): List<ForeignResourceCount>
    suspend fun shutdownView(): ForeignOwnerClose
    suspend fun start(): ForeignOwnerStart
    fun beginConnect(context: ForeignOwnerContext, peripheral: PeripheralId): ConnectOperation
    fun beginDisableNotifications(context: ForeignOwnerContext, connection: ConnectionId, session: GattSessionId, subscription: SubscriptionId): DisableNotificationsOperation
    fun beginDisconnect(context: ForeignOwnerContext, connection: ConnectionId): DisconnectOperation
    fun beginDiscoverGatt(context: ForeignOwnerContext, connection: ConnectionId): DiscoverGattOperation
    fun beginEnableNotifications(context: ForeignOwnerContext, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId, mode: NotificationModeRequest): EnableNotificationsOperation
    fun beginReadCharacteristic(context: ForeignOwnerContext, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId): ReadOperation
    fun beginReadRssi(context: ForeignOwnerContext, connection: ConnectionId): ReadRssiOperation
    fun beginScan(context: ForeignOwnerContext, options: ScanOptions): SupervisedScan
    fun beginStartScan(context: ForeignOwnerContext, options: ScanOptions): StartScanOperation
    fun beginStopScan(context: ForeignOwnerContext, session: ScanSessionId): StopScanOperation
    fun beginWriteCharacteristic(context: ForeignOwnerContext, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId, value: ByteArray, writeType: WriteType): WriteOperation
    fun beginScanView(context: ForeignOwnerContext, options: ForeignScanOptions): ForeignScanBeginResult
}
