pub enum ForeignPlatformRecoveryAction {
    WaitForRadioOn,
    AwaitOldOperationTerminals {remaining: Vec<String>,},
    AttemptSameOwnerRecovery {after_central_recreation: bool, target: ForeignRecoveryTarget,},
    RecreateCentral,
    AttemptFreshOwnerRecovery {target: ForeignRecoveryTarget,},

pub enum ForeignRecoveryAction {
    WaitForRadioOn,
    AwaitOldOperationTerminals {remaining: Vec<String>},
    AttemptSameOwnerRecovery {after_central_recreation: bool},
    RecreateCentral,
    RetireOldOwner,
    CreateFreshOwner,
    AttemptFreshOwnerRecovery,
}

pub struct ForeignRecoveryLink {
    pub connection_id: String,
    pub gatt_session_id: String,
    pub subscription_id: String,
}

pub enum ForeignRecoveryMode {
    SameOwner,
    FreshOwner,
}

pub struct ForeignRecoveryObservation {
    pub source_context: ForeignOwnerContext,
    pub signal: ForeignRecoverySignal,
}

pub enum ForeignRecoveryOutcome {
    Recovered {context: ForeignOwnerContext, mode: ForeignRecoveryMode, link: ForeignRecoveryLink,},
    PlatformTimeout {context: ForeignOwnerContext, reason: String,},
    ManualReconnectRequired {context: ForeignOwnerContext, reason: String,},
}

pub struct ForeignRecoveryPlan {
    pub execution_context: ForeignOwnerContext,
    pub sequence: u64,
    pub actions: Vec<ForeignRecoveryAction>,
    pub outcome: Option<ForeignRecoveryOutcome>,
}

pub struct ForeignRecoveryPolicyConfig {
    pub initial_context: ForeignOwnerContext,
    pub old_link: Option<ForeignRecoveryLink>,
    pub expected_old_operation_ids: Vec<String>,
    pub allow_central_recreation: bool,
    pub allow_fresh_owner: bool,
}

pub enum ForeignRecoverySignal {
    RadioLost,
    OldOperationTerminal {operation_id: String},
    RadioOn,
    SameOwnerRecovered {link: ForeignRecoveryLink},
    SameOwnerRecoveryFailed {reason: String},
    CentralRecreated,
    CentralRecreationFailed {reason: String},
    OwnerRetired {resources_zero: bool},
    FreshOwnerCreated,
    FreshOwnerRecovered {link: ForeignRecoveryLink},
    FreshOwnerRecoveryFailed {reason: String},
    PlatformTimedOut {reason: String},
    ManualReconnectRequired {reason: String},
}

pub struct SupervisedScan;

    pub async fn wait_until_started(&self, timeout_ms: u64) -> Result<(), ScanSessionError>;
    pub async fn cancel_start(&self) -> Result<(), ScanSessionError>;
    pub async fn next_event(&self, timeout_ms: u64) -> Result<ScanEvent, ScanSessionError>;
    pub fn begin_stop(&self) -> Result<Arc<StopScanOperation>, ScanSessionError>;
    pub fn detach(&self);
    pub fn observation_snapshot(&self) -> ForeignScanDeliverySnapshot;
    pub async fn start_result(&self, timeout_ms: f64) -> ForeignScanCommandResult;
    pub async fn cancel_start_result(&self) -> ForeignScanCommandResult;
    pub async fn next_observation(&self, timeout_ms: f64) -> ForeignScanObservation;
    pub fn begin_stop_view(&self) -> ForeignScanStopResult;
}

pub struct SupervisedScanStop;

impl SupervisedScanStop {
    pub async fn wait_result(&self, timeout_ms: f64) -> ForeignScanCommandResult;
    pub async fn cancel_result(&self) -> ForeignScanCommandResult;
}

pub struct Supervisor;

impl Supervisor {
    pub fn new(factory: Arc<dyn ForeignPlatformOwnerSessionFactory>, config: ForeignSupervisorConfig,) -> Result<Arc<Self>, SupervisorFfiError>;
    pub fn is_active(&self) -> Result<bool, SupervisorFfiError>;
    pub fn active_context(&self) -> Result<Option<ForeignOwnerContext>, SupervisorFfiError>;
    pub fn active_owner_start(&self) -> Result<Option<ForeignOwnerStart>, SupervisorFfiError>;
    pub async fn start(&self) -> Result<ForeignOwnerStart, SupervisorFfiError>;
    pub fn last_retired_resources(&self,) -> Result<Option<Vec<ForeignResourceCount>>, SupervisorFfiError>;
    pub fn configure_old_operation_barrier(&self, old_link: Option<ForeignRecoveryLink>, expected_old_operation_ids: Vec<String>,) -> Result<(), SupervisorFfiError>;
    pub async fn execute_platform_action(&self) -> Result<ForeignRecoveryPlan, SupervisorFfiError>;
    pub async fn drive_platform_actions_until_boundary(&self,) -> Result<ForeignRecoveryPlan, SupervisorFfiError>;
    pub async fn retire_for_recovery(&self) -> Result<ForeignRecoveryPlan, SupervisorFfiError>;
    pub async fn create_fresh_owner(&self, context: ForeignOwnerContext,) -> Result<ForeignRecoveryPlan, SupervisorFfiError>;
    pub async fn shutdown(&self) -> Result<Vec<ForeignResourceCount>, SupervisorFfiError>;
    pub async fn shutdown_view(&self) -> ForeignOwnerClose;
    pub fn observe(&self, observation: ForeignRecoveryObservation,) -> Result<ForeignRecoveryPlan, SupervisorFfiError>;
    pub fn observe_external_signal(&self, signal: ForeignExternalRecoverySignal,) -> Result<ForeignRecoveryPlan, SupervisorFfiError>;
    pub fn observe_signal(&self, signal: ForeignRecoverySignal,) -> Result<ForeignRecoveryPlan, SupervisorFfiError>;
    pub fn begin_scan(self: Arc<Self>, context: ForeignOwnerContext, options: ScanOptions,) -> Result<Arc<SupervisedScan>, ScanSessionError>;
    pub fn begin_start_scan(&self, context: ForeignOwnerContext, options: ScanOptions,) -> Result<Arc<StartScanOperation>, SupervisorOperationError>;
    pub fn begin_stop_scan(&self, context: ForeignOwnerContext, session: ScanSessionId,) -> Result<Arc<StopScanOperation>, SupervisorOperationError>;
    pub fn begin_connect(&self, context: ForeignOwnerContext, peripheral: PeripheralId,) -> Result<Arc<ConnectOperation>, SupervisorOperationError>;
    pub fn begin_disconnect(&self, context: ForeignOwnerContext, connection: ConnectionId,) -> Result<Arc<DisconnectOperation>, SupervisorOperationError>;
    pub fn begin_discover_gatt(&self, context: ForeignOwnerContext, connection: ConnectionId,) -> Result<Arc<DiscoverGattOperation>, SupervisorOperationError>;
    pub fn begin_read_characteristic(&self, context: ForeignOwnerContext, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId,) -> Result<Arc<ReadOperation>, SupervisorOperationError>;
    pub fn begin_write_characteristic(&self, context: ForeignOwnerContext, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId, value: Vec<u8>, write_type: WriteType,) -> Result<Arc<WriteOperation>, SupervisorOperationError>;
    pub fn begin_enable_notifications(&self, context: ForeignOwnerContext, connection: ConnectionId, session: GattSessionId, characteristic: GattCharacteristicId, mode: NotificationModeRequest,) -> Result<Arc<EnableNotificationsOperation>, SupervisorOperationError>;
    pub fn begin_disable_notifications(&self, context: ForeignOwnerContext, connection: ConnectionId, session: GattSessionId, subscription: SubscriptionId,) -> Result<Arc<DisableNotificationsOperation>, SupervisorOperationError>;
    pub fn begin_read_rssi(&self, context: ForeignOwnerContext, connection: ConnectionId,) -> Result<Arc<ReadRssiOperation>, SupervisorOperationError>;
    pub fn begin_scan_view(self: Arc<Self>, context: ForeignOwnerContext, options: ForeignScanOptions,) -> ForeignScanBeginResult;
}

    InvalidConfiguration {detail: String},
    NoActiveContext,
    Platform {detail: String},
}

pub enum SupervisorOperationError {
    NoActiveSession,
    StaleContext,
    RecoveryNotReady,
    SupervisorBusy,
    SupervisorUnavailable,
    InvalidContext {detail: String},
    Bleng {error: BlengError,},
}
